# PR #4 충돌 해결 - 수동 명령어

# 1. 원격 정보 가져오기
git fetch origin

# 2. PR 브랜치 확인 (GitHub CLI)
gh pr view 4 --json headRefName -q '.headRefName'

# 3. PR 브랜치로 체크아웃 (브랜치 이름은 위에서 확인한 값으로 변경)
git fetch origin <PR_브랜치_이름>:pr4-resolve
git checkout pr4-resolve

# 4. main 브랜치 merge
git fetch origin main
git merge origin/main

# 5. 충돌 해결 (자동으로 ours 사용 - PR 버전 유지)
git checkout --ours system/nginx_signalcraft_config.conf  # 또는 nginx_signalcraft_config.conf
git add system/nginx_signalcraft_config.conf

# 6. 커밋
git commit -m "충돌 해결: main 브랜치와 병합"

# 7. 푸시
git push origin pr4-resolve:<PR_브랜치_이름> --force-with-lease
