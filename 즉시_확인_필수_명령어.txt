프로덕션 서버에서 실행:

# 1. 서버가 정말로 3000번 포트에서 실행 중인지 확인
netstat -tlnp | grep 3000

# 2. curl로 localhost 테스트
curl http://localhost:3000/dashboard

# 3. curl로 상세 테스트
curl -v http://localhost:3000/dashboard 2>&1 | head -30

# 4. 서버 정보 확인
pm2 info signalcraft-app

# 5. 프로세스가 실행 중인 디렉토리 확인
pm2 info signalcraft-app | grep "cwd"

# 6. 실제 파일 경로로 이동해서 확인
cd $(pm2 info signalcraft-app | grep "cwd" | awk '{print $4}')
pwd
ls -la templates/customer/dashboard.html

# 7. server/app.js 파일 확인
grep -A 5 "app.get('/dashboard'" server/app.js

# 8. 실제 경로로 테스트
node -e "const path = require('path'); console.log(path.join(__dirname, '../templates/customer/dashboard.html'))" | head -1
ls -la "$(node -e "const path = require('path'); console.log(path.join(process.cwd(), 'templates/customer/dashboard.html'))")"

# 9. 서버의 모든 라우트 출력
pm2 logs signalcraft-app --lines 200 | grep -i "route\|dashboard"

# 10. 서버 재시작 후 즉시 로그 확인
pm2 restart signalcraft-app
sleep 2
pm2 logs signalcraft-app --lines 50
